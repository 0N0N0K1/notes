避免 SQL 注入的一般原则： 
1. 不信任用户提交的数据
2. 可以进行验证来测试用户的输入是否为已知安全的、已批准、已定义的输入
3. 使用用户的输入时，用  database/sql 的参数占位符来构造 SQL 语句，仅将它们作为自动转义参数
> [!warning] 注意:  Logger 打印的 SQL 并不像最终执行的 SQL 那样已经转义，复制和运行这些 SQL 时应当注意


> [!tip] 使用 ? 占位转义
```go
db.AutoMigrate(&TableGormStudy{})  

userInput := " '0N0N0K1';drop table table_gorm_studies;"  
// 安全的，会被转义  
db.Model(&TableGormStudy{}).Where("name = ?", userInput).Find(nil)  
//SELECT * FROM `table_gorm_studies` WHERE name = ' ''jinzhu'';drop table table_gorm_studies;'

// SQL 注入  
db.Model(&TableGormStudy{}).Where(fmt.Sprintf("name = %v", userInput)).Find(nil)
//SELECT * FROM `table_gorm_studies` WHERE name =  'jinzhu';drop table table_gorm_studies;

 // 安全的，会被转义 
db.Model(&TableGormStudy{}).Find(nil, "name = ?", userInput)  
//SELECT * FROM `table_gorm_studies` WHERE name = ' ''0N0N0K1'';drop table table_gorm_studies;'

// SQL 注入  
db.Model(&TableGormStudy{}).Find(nil, fmt.Sprintf("name = %v", userInput))
//SELECT * FROM `table_gorm_studies` WHERE name =  '0N0N0K1';drop table table_gorm_studies;
```

> [!tip] 使用前先检查是否为对应类型
```go
userInputID := "1=1;drop table users;"  
// safe, return error  
id, err := strconv.Atoi(userInputID)  
if err != nil {  
    return err  
}  
db.First(&user, id)  
  
// SQL injection  
db.First(&user, userInputID)  
// SELECT * FROM users WHERE 1=1;drop table users;
```


> [!tip] 为了支持某些功能，一些输入不会被转义，调用方法时要小心用户输入的参数
```go
db.Select("name; drop table users;").First(&user)  
db.Distinct("name; drop table users;").First(&user)  
  
db.Model(&user).Pluck("name; drop table users;", &names)  
  
db.Group("name; drop table users;").First(&user)  
  
db.Group("name").Having("1 = 1;drop table users;").First(&user)  
  
db.Raw("select name from users; drop table users;").First(&user)  
  
db.Exec("select name from users; drop table users;")  
  
db.Order("name; drop table users;").First(&user)  
  
db.Table("users; drop table users;").Find(&users)  
  
db.Delete(&User{}, "id=1; drop table users;")  
  
db.Joins("inner join orders; drop table users;").Find(&users)  
  
db.InnerJoins("inner join orders; drop table users;").Find(&users)  
  
//Despite being parameterized in Exec() function, gorm.Expr is still injectable  
db.Exec("UPDATE users SET name = '?' WHERE id = 1", gorm.Expr("alice'; drop table users;-- "))  
  
db.Where("id=1").Not("name = 'alice'; drop table users;").Find(&users)  
  
db.Where("id=1").Or("name = 'alice'; drop table users;").Find(&users)  
  
db.Find(&User{}, "name = 'alice'; drop table users;")  
  
// The following functions can only be injected by blind SQL injection methods  
db.First(&users, "2 or 1=1-- ")  
  
db.FirstOrCreate(&users, "2 or 1=1-- ")  
  
db.FirstOrInit(&users, "2 or 1=1-- ")  
  
db.Last(&users, "2 or 1=1-- ")  
  
db.Take(&users, "2 or 1=1-- ")
```
