>[!tip]
因为钩子方法的签名是固定的 func(*gorm.DB) error，没法往钩子里传参数
GORM 提供了 `Set`, `Get`, `InstanceSet`, `InstanceGet` 方法来允许用户传值给勾子 或 其他方法

# 1. 传递设置值到钩子方法
Set / Get ：
```go
type User struct {  
  gorm.Model  
  CreditCard CreditCard  
  // ...  
}    
type CreditCard struct {  
  gorm.Model  
  // ...  
}  
myValue := 123  
db.Set("my_value", myValue).Create(&User{})

func (u *User) BeforeCreate(tx *gorm.DB) error {  
  myValue, ok := tx.Get("my_value")  
  // ok => true  
  // myValue => 123 
}  
func (card *CreditCard) BeforeCreate(tx *gorm.DB) error {  
  myValue, ok := tx.Get("my_value")  
  // ok => true  
  // myValue => 123  
}  
  
```

InstanceSet / InstanceGet：
```go
type User struct {  
  gorm.Model  
  CreditCard CreditCard  
  // ...  
}  
type CreditCard struct {  
  gorm.Model  
  // ...  
} 
myValue := 123  
db.InstanceSet("my_value", myValue).Create(&User{})

func (u *User) BeforeCreate(tx *gorm.DB) error {  
  myValue, ok := tx.InstanceGet("my_value")  
  // ok => true  
  // myValue => 123  
}  
// 在创建关联时，GORM 创建了一个新 `*Statement`，所以它不能读取到其它实例的设置  
func (card *CreditCard) BeforeCreate(tx *gorm.DB) error {  
  myValue, ok := tx.InstanceGet("my_value")  
  // ok => false  
  // myValue => nil  
}  
```
使用
1. 自动填充字段
```go
// 调用处
db.Set("operator_id", currentUser.ID).Create(&order)

// 模型钩子
func (o *Order) BeforeCreate(tx *gorm.DB) error {
    if v, ok := tx.Get("operator_id"); ok {
        operatorID := v.(uint)
        tx.Statement.SetColumn("CreatedBy", operatorID)
    }
    return nil
}
```
2. 跳过某些钩子逻辑
```go
// 批量导入时跳过"发通知"逻辑
db.Set("skip_notify", true).CreateInBatches(&users, 100)

func (u *User) AfterCreate(tx *gorm.DB) error {
    if v, ok := tx.Get("skip_notify"); ok && v.(bool) {
        return nil  // 跳过通知
    }
    return sendWelcomeNotification(u)
}
```

#  2. 迁移时传递表选项
```go
// 迁移时传递表选项（MySQL 引擎、字符集等）
db.Set("gorm:table_options", "ENGINE=InnoDB CHARSET=utf8mb4").AutoMigrate(&User{})
// 生成: CREATE TABLE `users` (...) ENGINE=InnoDB CHARSET=utf8mb4
```

# 3. 注意
1. panic
```go
//值不存在或类型不对时直接 panic
operatorID := tx.Get("operator_id").(uint)
// 安全的取法
if v, ok := tx.Get("operator_id"); ok {
    if id, ok := v.(uint); ok {
        // 使用 id
    }
}
```

2.  InstanceSet 在关联表的HOOK取不到
```go
// 以为关联模型的 Hook 也能拿到
db.InstanceSet("tenant_id", tenantID).Create(&user)
// User 的 Hook 能拿到，但 user.CreditCard 的 Hook 拿不到


//如果希望所有关联模型的 Hook 都能拿到，必须用 Set：
// 多租户 + 关联写入
db.Set("tenant_id", tenantID).Create(&user)  
// user 和关联的 CreditCard 都能读到
```

3. Set 的值在 Session 衍生链中一直存在
```go
db.Set("operator_id", 123)
// 之后同一 db 实例上的所有操作都带这个值
db.Create(&a)  // Hook 里能读到
db.Create(&b)  // Hook 里也能读到

//如果不希望"污染"后续操作，用新的 Session：
tx := db.Session(&gorm.Session{}).Set("operator_id", 123)
tx.Create(&a)  // 只有这个链里有
db.Create(&b)  // 读不到
```
