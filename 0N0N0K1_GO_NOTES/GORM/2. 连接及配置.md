```go
// 连接配置（全局）
conf := &gorm.Config{
	// 是否跳过默认事务（开启后 Create/Update/Delete 不再自动包裹事务）

    NoLowerCase: true, // skip the snake_casing of names

  },

	FullSaveAssociations: false,
	// 日志器，控制 SQL 日志的输出级别与格式

	NowFunc: func() time.Time {  
    return time.Now().Local()  
  }，
	// 只生成 SQL 不执行（用于调试/测试）
	DryRun: false,
	// 是否启用预编译语句（Prepared Statement）缓存复用
	PrepareStmt: false,
	// 预编译语句缓存的最大数量
	PrepareStmtMaxSize: 0,
	// 预编译语句缓存的存活时间
	PrepareStmtTTL: 0,
	// 是否禁用打开连接时的自动 Ping 检查
	DisableAutomaticPing: false,
	// 自动迁移时是否禁用外键约束创建
	DisableForeignKeyConstraintWhenMigrating: false,
	// 自动迁移时是否忽略模型间关系
	IgnoreRelationshipsWhenMigrating: false,
	// 是否禁用嵌套事务（SavePoint）
	DisableNestedTransaction: false,
	// 是否允许不带 WHERE 条件的全局 Update/Delete
	AllowGlobalUpdate: false,
	// 查询时 SELECT 显式列出字段名而不是 SELECT *
	QueryFields: false,
	// 批量 Create 时每批插入的条数
	CreateBatchSize: 0,
	// 是否把数据库驱动的错误翻译成 GORM 定义的通用错误（如主键冲突）
	TranslateError: false,
	// 是否将 Unscoped 传播到关联操作
	PropagateUnscoped: false,
	// 自定义 SQL 子句（Clause）构造器
	ClauseBuilders: nil,
	// 自定义连接池（传入 sql.DB 或包装后的连接池）
	ConnPool: nil,
	// 数据库方言（指定 MySQL/Postgres/SQLite 等驱动）
	Dialector: nil,
	// 注册的 GORM 插件列表
	Plugins: nil,
}
```

```go
// 连接地址
dsn := "root:006620@tcp(127.0.0.1:3306)/gorm_study?charset=utf8mb4&parseTime=True&loc=Local"  
//root:006620 —— MySQL 用户名和密码
//tcp(127.0.0.1:3306) —— 数据库地址和端口
//gorm_study —— 要连接的数据库名
//charset=utf8mb4 —— 字符集（支持 emoji 和完整 Unicode）
//parseTime=True —— 把数据库里的 DATETIME/TIMESTAMP 自动解析为 Go 的 time.Time
//loc=Local —— 时间按本地时区解析  
```

```go
// 连接
db, err := gorm.Open(mysql.Open(dsn), conf )  
if err != nil {  
    return nil, err  
}  
```
  
``` go
// 配置连接池  
sqlDB, err := db.DB()  
if err != nil {  
    return nil, err  
}  
sqlDB.SetMaxIdleConns(10)                  // 空闲连接数  
sqlDB.SetMaxOpenConns(100)                 // 最大连接数  
sqlDB.SetConnMaxLifetime(time.Hour)        // 连接最长存活时间  
sqlDB.SetConnMaxIdleTime(10 * time.Minute) // 连接最长空闲时间  
```

>[!tip]
>1. 得到的db为* gorm.DB，是全局连接与操作数据库的句柄，使用的是连接时的全局配置
>
>2. 会话 ,GORM 提供了 `Session` 方法，它允许创建一个带新配置的新建会话，会话配置与全局配置关系
>- 布尔型选项只能开启、不能关闭
>- 值类型的字段不受零值困扰，会话值直接覆盖全局值

```go
// Session 配置  
type Session struct {  
  DryRun                   bool  
  PrepareStmt              bool  
  NewDB                    bool  
  Initialized              bool  
  SkipHooks                bool  
  SkipDefaultTransaction   bool  
  DisableNestedTransaction bool  
  AllowGlobalUpdate        bool  
  FullSaveAssociations     bool  
  QueryFields              bool  
  Context                  context.Context  
  Logger                   logger.Interface  
  NowFunc                  func() time.Time  
  CreateBatchSize          int  
}
```

```go
// 新建会话模式  
stmt := db.Session(&Session{DryRun: falus}).First(&user, 1)
```
